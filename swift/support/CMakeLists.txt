target_sources(xplpc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/xplpc.h")
set_property(TARGET xplpc APPEND PROPERTY PUBLIC_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/xplpc.h")

set_target_properties(xplpc PROPERTIES
    FRAMEWORK TRUE
    FRAMEWORK_VERSION A
    MACOSX_FRAMEWORK_IDENTIFIER "com.xplpc.library"
    VERSION ${XPLPC_VERSION}
    SOVERSION ${XPLPC_VERSION_CODE}
    MACOSX_FRAMEWORK_BUNDLE_VERSION "${XPLPC_VERSION}"
    MACOSX_FRAMEWORK_SHORT_VERSION_STRING "${XPLPC_VERSION}"
    XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY "iPhone Developer"
    XCODE_ATTRIBUTE_ONLY_ACTIVE_ARCH YES
    XCODE_ATTRIBUTE_SKIP_INSTALL NO
    XCODE_ATTRIBUTE_CLANG_CXX_LIBRARY "libc++"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.xplpc.library"
    XCODE_ATTRIBUTE_DEFINES_MODULE YES
    XCODE_ATTRIBUTE_MODULEMAP_FILE "${CMAKE_CURRENT_SOURCE_DIR}/module.modulemap"
    XCODE_ATTRIBUTE_BUILD_LIBRARY_FOR_DISTRIBUTION YES
)
